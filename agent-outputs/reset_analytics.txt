// 4. PASSWORD RESET FUNCTION (global scope)
async function resetPassword(){
 const email = document.getElementById('email').value.trim();
 if(!email){
   document.getElementById('authErr').textContent = 'Enter your email above first, then tap Forgot Password';
   return;
 }
 try{
   await firebase.auth().sendPasswordResetEmail(email);
   document.getElementById('authErr').textContent = 'Check your email for a reset link.';
 }catch(e){
   if(e.code === 'auth/user-not-found'){
     document.getElementById('authErr').textContent = 'No account with that email.';
   }else{
     document.getElementById('authErr').textContent = 'Error sending password reset email.';
   }
 }
}

// 5. RETURN VISIT CHECK FUNCTION (global scope, called after loadUserData() succeeds)
async function checkReturnVisit(){
 const user = firebase.auth().currentUser;
 if(!user) return;
 const uid = user.uid;
 const today = new Date().toISOString().slice(0,10);
 const docRef = firebase.firestore().collection('users').doc(uid);
 try{
   const doc = await docRef.get();
   if(doc.exists){
     const data = doc.data();
     if(data.lastActive && data.lastActive !== today){
       analytics.logEvent('return_visit');
     }
   }
   await docRef.update({lastActive: today});
 }catch(e){
   // silently fail – analytics is best effort
 }
}
