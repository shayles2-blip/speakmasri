```js
let rehearsalIdx, rehearsalPack, currentMediaStream, currentRecorder, currentAudioURL, micAvailable = true;

function startRehearsal(){
  rehearsalPack = curLesson.vocab;
  rehearsalIdx = 0;
  document.getElementById("done").classList.add("hidden");
  document.getElementById("rehearsal").classList.remove("hidden");
  micAvailable = true;
  cleanupRecording();
  renderRehearsalItem();
}

function renderRehearsalItem(){
  let item = rehearsalPack[rehearsalIdx];
  let content = document.getElementById("rehearsal-content");
  let html = `
    <div style="direction: rtl; font-size: 2em; margin-bottom: 10px;">${item.ar}</div>
    <div>${item.franco}</div>
    <div>${item.en}</div>
    <div>${registerBadge(item)}</div>
    <button id="hearBtn">🔊 Hear it</button>
    ${micAvailable ? `<button id="recordBtn">Hold to Record</button>` : `<div>Mic access not available – just practice saying it out loud</div>`}
    <button id="nextBtn">${rehearsalIdx === rehearsalPack.length - 1 ? 'Finish' : 'Next'}</button>
  `;
  content.innerHTML = html;

  document.getElementById("hearBtn").onclick = () => speak(item.ar);

  if(micAvailable) {
    let recordBtn = document.getElementById("recordBtn");
    setupRecordButton(recordBtn);
  }

  document.getElementById("nextBtn").onclick = () => {
    cleanupRecording();
    rehearsalIdx++;
    if(rehearsalIdx >= rehearsalPack.length){
      content.innerHTML = `<h2>All done! 🎉</h2><p>You've practised the whole lesson.</p><button id="finishRehearsalBtn">Back to lessons</button>`;
      document.getElementById("finishRehearsalBtn").onclick = () => {
        document.getElementById("rehearsal").classList.add("hidden");
        cleanupRecording();
        finishLesson();
      };
    } else {
      renderRehearsalItem();
    }
  };
}

function setupRecordButton(btn){
  btn.addEventListener('mousedown', startRecording);
  btn.addEventListener('touchstart', startRecording, {passive: false});
  btn.addEventListener('mouseup', stopRecording);
  btn.addEventListener('touchend', stopRecording);
  btn.addEventListener('mouseleave', stopRecording);
}

function startRecording(e){
  e.preventDefault();
  let btn = e.currentTarget;
  btn.textContent = 'Recording...';
  btn.style.backgroundColor = 'red';
  navigator.mediaDevices.getUserMedia({audio: true})
    .then(stream => {
      currentMediaStream = stream;
      currentRecorder = new MediaRecorder(stream);
      let chunks = [];
      currentRecorder.ondataavailable = e => chunks.push(e.data);
      currentRecorder.onstop = () => {
        let blob = new Blob(chunks, {type: 'audio/webm'});
        if(currentAudioURL) URL.revokeObjectURL(currentAudioURL);
        currentAudioURL = URL.createObjectURL(blob);
        let audio = document.getElementById("rehearsalPlayback");
        audio.src = currentAudioURL;
        audio.play();
      };
      currentRecorder.start();
    })
    .catch(err => {
      console.error(err);
      micAvailable = false;
      btn.removeEventListener('mousedown', startRecording);
      btn.removeEventListener('touchstart', startRecording);
      renderRehearsalItem(); // show fallback message
    });
}

function stopRecording(e){
  if(currentRecorder && currentRecorder.state === 'recording'){
    currentRecorder.stop();
  }
  if(currentMediaStream){
    currentMediaStream.getTracks().forEach(track => track.stop());
  }
  let btn = document.getElementById("recordBtn");
  if(btn){
    btn.textContent = 'Hold to Record';
    btn.style.backgroundColor = '';
  }
}

function cleanupRecording(){
  if(currentRecorder && currentRecorder.state === 'recording'){
    currentRecorder.stop();
  }
  if(currentMediaStream){
    currentMediaStream.getTracks().forEach(track => track.stop());
    currentMediaStream = null;
  }
  if(currentAudioURL){
    URL.revokeObjectURL(currentAudioURL);
    currentAudioURL = null;
  }
  currentRecorder = null;
}
```
